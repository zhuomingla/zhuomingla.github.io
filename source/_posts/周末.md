 ## 矩阵
 最讨厌的就是分不清究竟是哪行哪列，当然，也可能是我没怎么练习……  
$$
    A=
    \begin{vmatrix}
    a & b \\
    c & d
    \end{vmatrix}
    $$
$A_{12} = b$ 先行后列……  
可我是真的一行一行数的。算了  
用矩阵表示线性方程组，行数代表方程数，列数表示变量加结果。称为增广矩阵?  
可进行的操作：
- 交换行
- 行乘以常数
- 两行相加到其中一行

就是求解方程组操作的映射。  
由于每列代表一个变量，那么除输出外其他每列只有一行有参数且互不在同一列就可以直接读出变量值  
$$
    A=
    \begin{vmatrix}
    1 & 0 & 2\\
    0 & 1 & 3
    \end{vmatrix}
    $$
x = 2, y = 3  

加减很容易理解，乘法  
$点乘积=> (a,b) *(c,d)= （ac+bd）$

A*B = C,$C_{1,2} 即A行和B列的点乘积$  
为什么要这么定义？  
怎么理解呢？  
线性变化：
- 一个对象可以拆成多个同类型对象的加减组合
- 一个对象乘常量可以等同于其拆分组合乘常量
- 一个对象的映射等同于其组合的映射

一个向量（1，2）可以拆分成x和y基向量的组合，同时变换可以看成映射，为了求得向量的映射，可以看基向量的映射。  
$$
    A=
    \begin{vmatrix}
    1 & 2 \\
    3 & 4
    \end{vmatrix}
    $$

（1，2）= 1 * （1，0）+ 2* （0，1）  
（？，？） = 1 * （1，3） + 2 * （2，4）
这一正好与把矩阵放左边与向量相乘的运算相同……  
感觉我是不是弄错了什么……
继续简化的说，变换就是找到子组合在变化后的对象在重新组合成原对象？  
这样来推导一下顺时针旋转。  
p = x （1，0）+ y （0，1）  
顺时针旋转j度后  
（1，0）-》$（cos(j),-sin(j)）$  
（0，1）-》$ (sin(j),cos(j)) $  
然后组成，，，应该差不多？  
因此，要明白一个对象怎么变化，就只要关注其子对象的变化，再重组就可以了？  我应该上个学期就学完了吧……

  var world   = new Uint8Array(WW * WH);
  var heights = new Int16Array(WW);

  function getT(x, y) {
    if (x < 0 || x >= WW || y < 0 || y >= WH) return STONE;
    return world[y * WW + x];
  }
  function setT(x, y, v) {
    if (x < 0 || x >= WW || y < 0 || y >= WH) return;
    world[y * WW + x] = v;
  }
  function solid(x, y) {
    if (x < 0 || x >= WW || y < 0 || y >= WH) return true;
    return world[y * WW + x] !== AIR;
  }

  // ==================== 噪声 ====================
  function hash2(x, y) {
    var n = Math.sin(x * 127.1 + y * 311.7) * 43758.5453;
    return n - Math.floor(n);
  }
  function vnoise(x, y) {
    var ix = Math.floor(x), iy = Math.floor(y);
    var fx = x - ix, fy = y - iy;
    var ux = fx * fx * (3 - 2 * fx);
    var uy = fy * fy * (3 - 2 * fy);
    var a = hash2(ix, iy),     b = hash2(ix + 1, iy);
    var c = hash2(ix, iy + 1), d = hash2(ix + 1, iy + 1);
    return (a * (1 - ux) + b * ux) * (1 - uy) +
           (c * (1 - ux) + d * ux) * uy;
  }
  function fbm(x, y, oct) {
    var v = 0, amp = 1, f = 1, tot = 0;
    for (var i = 0; i < oct; i++) {
      v += vnoise(x * f, y * f) * amp;
      tot += amp; amp *= 0.5; f *= 2;
    }
    return v / tot;
  }

  function generate() {
    var x, y, i, dx, dy;

    // 地表高度

      var n = fbm(x * 0.02, 0.5, 4);
      var h = Math.round(60 + (n - 0.5) * 40);
      if (h < 25) h = 25;

          }

        // 树干

  // ==================== 主更新 ====================
  function update() {
    var d = 0;
    if (keys.left)  d--;
    if (keys.right) d++;

    if (d !== 0) {
      player.vx += d * ACC;
      player.face = d;
    } else {
      player.vx *= FRIC;
      if (Math.abs(player.vx) < 0.05) player.vx = 0;
    }
    if (player.vx >  MAXV) player.vx =  MAXV;
    if (player.vx < -MAXV) player.vx = -MAXV;

    if (keys.jump && player.ground) {
      player.vy = JUMP;
      player.ground = false;
    }

    player.vy += GRAV;
    if (player.vy > MAXFALL) player.vy = MAXFALL;

    player.x += player.vx;  moveX();
    player.ground = false;
    player.y += player.vy;  moveY();

    updateCamera(false);
    updateActions();
    updateParts();
  }

  // ==================== 循环 ====================
  var last = performance.now();
  var acc = 0;
  var STEP = 1000 / 60;

  function loop(now) {
    requestAnimationFrame(loop);

    var dt = now - last;
    last = now;
    if (dt > 200) dt = 200;

    acc += dt;
    var guard = 0;
    while (acc >= STEP && guard < 5) {
      update();
      acc -= STEP;
      guard++;
    }
    draw();
  }

  // ==================== 启动 ====================
  generate();

  var sx = Math.floor(WW / 2);
  player.x = sx * TILE + (TILE - player.w) / 2;
  player.y = (heights[sx] - 5) * TILE;

  updateCamera(true);
  requestAnimationFrame(loop);

})();
